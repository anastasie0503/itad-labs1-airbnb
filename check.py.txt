import duckdb

con = duckdb.connect()
con.execute("INSTALL httpfs; LOAD httpfs;")
con.execute("""
CREATE SECRET seaweed (
    TYPE s3,
    KEY_ID 's3admin',
    SECRET 's3admin123',
    ENDPOINT 'seaweedfs:8333',
    URL_STYLE 'path',
    USE_SSL false
);
""")

print("Читаю данные из SeaweedFS...")
res = con.execute("SELECT id, name, price FROM read_csv_auto('s3://raw/nyc-airbnb-listings/ingested_on=2026-11-03/listings.csv') LIMIT 5").fetchdf()
print(res)
